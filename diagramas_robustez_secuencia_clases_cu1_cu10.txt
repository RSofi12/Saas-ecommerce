================================================================================
PARTE 1: GUIA DE USO EN ENTERPRISE ARCHITECT
================================================================================
A. PARA IMPORTAR LOS DIAGRAMAS DE ROBUSTEZ Y SECUENCIA (XMI):
   1. Guarda la Seccion 2 en un archivo con extension '.xmi' o '.xml' (ej: 'cu01_cu10_robustez_secuencia.xmi').
   2. Abre Enterprise Architect.
   3. Haz clic derecho sobre el paquete de destino en el Project Browser > 'Import/Export' > 'Import Package from XMI File...'.
   4. Selecciona el archivo XMI y pulsa 'Import'. Se generaran los paquetes con los actores, boundaries, controllers, entities y diagramas de secuencia.

B. PARA GENERAR EL DIAGRAMA DE CLASES RELACIONAL (JSCRIPT):
   1. En Enterprise Architect, presiona Ctrl+9 (o menu Tools / Add-Ins > Manage > Script).
   2. En la ventana 'Scripting', haz clic en 'New JScript'.
   3. Pega el codigo de la PARTE 3 de este documento.
   4. Selecciona un paquete en el Project Browser y presiona 'Run' (F5).
   5. Se creara el paquete 'Modelo de Datos FashionStore SaaS' con las 39 tablas, atributos tipados, PK, FK y 72 asociaciones.

function eaType(sqlType) {
    var t = (sqlType + "").toLowerCase();
    if (t.indexOf("uuid") >= 0) return "Guid";
    if (t.indexOf("boolean") >= 0) return "Boolean";
    if (t.indexOf("integer") >= 0 || t.indexOf("serial") >= 0) return "Integer";
    if (t.indexOf("numeric") >= 0 || t.indexOf("decimal") >= 0 || t.indexOf("money") >= 0) return "Currency";
    if (t.indexOf("timestamp") >= 0 || t.indexOf("date") >= 0) return "Date";
    if (t.indexOf("json") >= 0) return "String";
    if (t.indexOf("text") >= 0 || t.indexOf("character") >= 0 || t.indexOf("char") >= 0 || t.indexOf("varchar") >= 0) return "String";
    return "String";
}

function main() {
    var targetPkg = Repo.GetTreeSelectedPackage();
    if (targetPkg == null) {
        targetPkg = Repo.Models.GetAt(0);
    }

    Log("=== Iniciando creacion del Modelo de Datos FashionStore SaaS (39 Tablas) ===");

    var pkg = targetPkg.Packages.AddNew("Modelo de Datos FashionStore SaaS", "");
    pkg.Update();

    var classMap = {};

    for (var i = 0; i < TABLAS.length; i++) {
        var tName = TABLAS[i][0];
        var cols  = TABLAS[i][1];

        var el = pkg.Elements.AddNew(tName, "Class");
        el.Stereotype = "table";
        el.Update();
        classMap[tName] = el;

        for (var c = 0; c < cols.length; c++) {
            var colName  = cols[c][0];
            var colSql   = cols[c][1];
            var colStereo= cols[c][2];

            var attr = el.Attributes.AddNew(colName, eaType(colSql));
            if (colStereo != null) {
                attr.Stereotype = colStereo;
            }
            attr.Update();
        }
        el.Attributes.Refresh();
    }
    pkg.Elements.Refresh();
    Log("Tablas creadas: " + TABLAS.length);

    var relCount = 0;
    for (var r = 0; r < RELACIONES.length; r++) {
        var oName  = RELACIONES[r][0];
        var dName  = RELACIONES[r][1];
        var verbo  = RELACIONES[r][2];
        var cardO  = RELACIONES[r][3];
        var cardD  = RELACIONES[r][4];

        var oEl = classMap[oName];
        var dEl = classMap[dName];

        if (oEl != null && dEl != null) {
            var conn = oEl.Connectors.AddNew(verbo, "Association");
            conn.ClientID = oEl.ElementID;
            conn.SupplierID = dEl.ElementID;
            conn.ClientEnd.Cardinality = cardO;
            conn.SupplierEnd.Cardinality = cardD;
            conn.Update();
            oEl.Connectors.Refresh();
            relCount++;
        }
    }
    Log("Relaciones creadas: " + relCount);

    var diag = pkg.Diagrams.AddNew("Diagrama de Clases - Modelo Relacional", "Class");
    diag.Update();

    for (var k in classMap) {
        var dObj = diag.DiagramObjects.AddNew("", "");
        dObj.ElementID = classMap[k].ElementID;
        dObj.Update();
    }
    diag.DiagramObjects.Refresh();
    diag.Update();

    pkg.Packages.Refresh();
    Repo.RefreshModelView(pkg.PackageID);
    Log("=== Modelo de Datos creado exitosamente en Enterprise Architect ===");
}

main();
